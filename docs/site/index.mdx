import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="The integration layer" icon="puzzle">
    [MTL5](https://stillwater-sc.github.io/mtl5/) is the linear-algebra layer and
    does not depend on Universal. mp-numerics is where its algorithms meet
    [Universal](https://stillwater-sc.github.io/universal/)'s number types.
  </Card>
  <Card title="Measured exactly" icon="approve-check">
    Computed values are read out of the number type as exact expansions, so a
    reported error is the error of the arithmetic, not of the measurement.
  </Card>
  <Card title="Reproducible" icon="document">
    Every committed result carries a `.sysinfo` sidecar with the platform,
    compiler and the commits of mp-numerics, MTL5 and Universal it was run
    against.
  </Card>
  <Card title="Organized by method" icon="list-format">
    Quadrature, root finding, interpolation, optimization, approximation and
    linear algebra, each with applications, benchmarks and tests.
  </Card>
</CardGrid>
